```cpp
    uint8_t width;
    uint8_t height;
    int8_t x_offset;

    int16_t screen_x;
    int16_t screen_y;
    int16_t width;
    int16_t height;
    int16_t next_sprite;
    int16_t layer;      // 추정: 렌더 레이어

    int16_t sprite_ptr;    // +0x04
    int16_t x, y;          // +0x06, +0x08

};

struct Camera {
    int16_t x, y;
};

int setup_sprite_array2(Object* obj, Camera& camera) {
    // 1. 스프라이트 포인터 초기화
    if (obj->sprite_ptr == -1) {
        obj->sprite_ptr = calculate_sprite_pointer(obj);
    }

    // 2. 연결 리스트 다음 노드
    SpriteNode* sprite = *(SpriteNode**)obj->sprite_ptr;

    if (sprite == (SpriteNode*)0xffff) {
        return 0;  // 리스트 끝
    }

    // 3. 렌더 데이터 설정
    RenderData* out = obj->output;

    out->next_sprite = sprite->next;
    out->width = sprite->width;
    out->height = sprite->height;

    // 4. 화면 좌표 계산
    out->screen_x = (sprite->x_offset + obj->x) - camera.x;

    // Y 좌표 변환 (타일 → 픽셀)
    int tile_y = obj->y - camera.y;
    out->screen_y = sprite->y_offset + (tile_y * 4);

    // 5. 추가 데이터
    out->layer = g_render_layer;  // 0xf4f0

    return out->screen_y;
}

        Object* obj = &projectile_array[i];

        }
    }
```

### 1. Y 좌표 스케일 팩터 (× 4)

**원작**:
```c

```

**의미**:
- CGA 모드 4: 4픽셀 높이 타일
- 또는 논리 좌표 → 물리 좌표 변환
- 스프라이트 정렬 기준

### 2. 연결 리스트 vs 배열

**원작**: 연결 리스트
- 스프라이트 프레임 간 순회
- 애니메이션 시퀀스

**현대**: 배열 기반
```cpp
struct SpriteSheet {
    std::vector<SpriteFrame> frames;
    int current_frame = 0;

    const SpriteFrame& get_current() const {
        return frames[current_frame];
    }

    void next_frame() {
        current_frame = (current_frame + 1) % frames.size();
    }
};
```

### 3. 포인터의 포인터

**원작**:

**의미**: 간접 참조로 리스트 순회 상태 유지

**현대**:
```cpp
class SpriteIterator {
    SpriteNode** current;

public:
    SpriteNode* next() {
        SpriteNode* node = *current;
        if (node && node->next != -1) {
            *current = &node->next;
        }
        return node;
    }
};
```
