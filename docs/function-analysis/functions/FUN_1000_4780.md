## 💡 핵심 발견

### 1. 컨티뉴 시스템

**컨티뉴 횟수**: 5회 (0x442c = 5)

- 0이 되면 타이틀로

**플레이어 목숨**: 각 2개
- 0x4428: P1 lives
- 0x442a: P2 lives

### 2. VGA 타이밍 동기화

**포트 0x3da**: VGA 상태 레지스터
```c
// 수직 리트레이스 대기 (frame sync)
while ((in(0x3da) & 8) == 0);  // 시작 대기
while ((in(0x3da) & 8) != 0);  // 종료 대기
```

→ CGA/EGA 시대 표준 vsync 기법

### 3. 게임 흐름 제어

```

    ↓
```

## 🔄 C++ 재구현

```cpp
enum class PlayerState : uint8_t {
    Dead = 0,
    Alive = 0x2a
};

struct GameState {
    int continues_left = 5;
    int p1_lives = 2;
    int p2_lives = 2;
    int current_level = 1;
    bool game_over = false;
};

class GameOverScreen {
public:
    void update(GameState& state, Player& p1, Player& p2) {
        // 1. 게임 오버 조건 체크
        if (p1.state != PlayerState::Dead ||
            p2.state != PlayerState::Dead) {
            return;  // 한 명이라도 살아있으면 무시
        }

        // 2. GAME OVER 화면 표시 (160 프레임)
        display_game_over_screen();
        wait_frames(160);

        // 3. 컨티뉴 프롬프트
        if (state.continues_left > 0) {
            if (show_continue_prompt()) {
                // 컨티뉴 선택
                state.continues_left--;
                respawn_players(p1, p2);
                return;
            }
        }

        // 4. 게임 완전 리셋
        reset_game(state, p1, p2);
    }

private:
    bool show_continue_prompt() {
        const int TIMEOUT_FRAMES = 360;  // 6초

        for (int frame = 0; frame < TIMEOUT_FRAMES; frame++) {
            // 입력 체크
            if (input.p1_button_pressed()) {
                return true;  // P1 컨티뉴
            }
            if (input.p2_button_pressed()) {
                return true;  // P2 컨티뉴
            }

            // 화면 업데이트
            update_continue_screen(TIMEOUT_FRAMES - frame);
            wait_vsync();
        }

        return false;  // 타임아웃
    }

    void respawn_players(Player& p1, Player& p2) {
        // 플레이어 리셋
        p1.state = PlayerState::Alive;
        p2.state = PlayerState::Alive;
        p1.lives = 2;
        p2.lives = 2;

        // 빈 공간 찾기
        auto spawn_pos = find_spawn_position();
        p1.x = spawn_pos.x;
        p1.y = spawn_pos.y;
        p2.x = spawn_pos.x + 20;
        p2.y = spawn_pos.y;
    }

    void reset_game(GameState& state, Player& p1, Player& p2) {
        // 컨티뉴 횟수 리셋
        state.continues_left = 5;

        // 플레이어 초기화
        p1.lives = 2;
        p2.lives = 2;
        p1.state = PlayerState::Alive;
        p2.state = PlayerState::Alive;

        // 레벨 1부터 시작
        state.current_level = 1;

        // 컬러 페이드 효과
        fade_effect(20);

        // 타이틀 화면으로
        state.game_over = true;
    }

    void wait_vsync() {
        // 원작: VGA port 0x3da 폴링
        // 현대: SDL_RenderPresent() or vsync
#ifdef SDL2
        SDL_RenderPresent(renderer);
#else
        // 60 FPS 기준
        std::this_thread::sleep_for(
            std::chrono::milliseconds(16)
        );
#endif
    }

    void wait_frames(int count) {
        for (int i = 0; i < count; i++) {
            wait_vsync();
        }
    }
};
```

```cpp
        switch (state) {
            case GameState::Playing:

                if (all_players_dead()) {
```
