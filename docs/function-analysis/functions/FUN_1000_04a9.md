```cpp
enum class AnimationFrame : uint8_t {
    None = 0,   // 애니메이션 없음
    Frame0 = 1,
    Frame1 = 2,
    Frame2 = 3
};

enum class ObjectState : uint8_t {
    Idle = 0x00,
    Walking = 0x01,
    Running = 0x02,
    // ...
    Special = 0x36
};

enum class SpecialFlags : uint8_t {
    None = 0,
    Jumping = 0x08,  // bit 3
};

struct ObjectFlags {
    AnimationFrame anim_frame : 2;  // bit 0-1
    uint8_t unused : 1;             // bit 2
    bool special : 1;               // bit 3
    uint8_t reserved : 4;           // bit 4-7

    AnimationFrame get_frame() const {
        return static_cast<AnimationFrame>(anim_frame);
    }

    bool has_special_flag() const {
        return special;
    }
};

struct Object {
    ObjectState state;      // +0x02
    ObjectFlags flags;      // +0x04
    int16_t sprite_ptr;     // +0x06
};

class SpriteSelector {
    // 데이터 테이블 (원본 바이너리에서 추출)
    static constexpr int16_t SPRITE_TABLE_0x1930[256] = {
        // ... 원본 데이터
    };

public:
    void select_sprite(Object* obj) {
        // 1. 애니메이션 프레임 체크
        AnimationFrame frame = obj->flags.get_frame();

        if (frame == AnimationFrame::None) {
            // 애니메이션 없음 → idle 또는 종료
            handle_no_animation(obj);
            return;
        }

        // 2. 테이블 인덱스 계산
        int anim_index = static_cast<int>(frame) - 1;  // 0, 1, 2
        int state_value = static_cast<int>(obj->state);
        int table_index = anim_index + state_value;

        // 3. 스프라이트 포인터 읽기
        int16_t sprite_ptr = SPRITE_TABLE_0x1930[table_index];

        // 4. 특수 상태 오프셋
        if (obj->flags.has_special_flag()) {
            if (obj->state == ObjectState::Running ||
                obj->state == ObjectState::Special) {
                sprite_ptr += 0x38;  // 특수 스프라이트 세트
            }
        }

        // 5. 저장
        obj->sprite_ptr = sprite_ptr;
    }

private:
    void handle_no_animation(Object* obj) {
        // idle 스프라이트 설정 또는
        // 객체 비활성화
        obj->sprite_ptr = -1;
    }
};

// 사용 예시
void update_character_sprite(Object* character) {
    sprite_selector.select_sprite(character);
}
```

```cpp
    std::vector<SpriteFrame> frames;
    int current_frame = 0;
    float frame_time = 0.0f;
    float frame_duration = 0.1f;  // 100ms per frame

public:
    void update(float dt) {
        frame_time += dt;
        if (frame_time >= frame_duration) {
            frame_time = 0.0f;
            current_frame = (current_frame + 1) % frames.size();
```

**원작**: 컴파일 타임 테이블

**현대**: std::map 또는 std::unordered_map
```cpp
class SpriteDatabase {
    struct SpriteKey {
        ObjectState state;
        AnimationFrame frame;
        bool special;

        bool operator==(const SpriteKey& other) const {
            return state == other.state &&
                   frame == other.frame &&
                   special == other.special;
        }
    };

    struct KeyHash {
        size_t operator()(const SpriteKey& key) const {
            return (static_cast<int>(key.state) << 4) |
                   (static_cast<int>(key.frame) << 2) |
                   (key.special ? 1 : 0);
        }
    };

    std::unordered_map<SpriteKey, SpriteSheet*, KeyHash> sprites;

public:
    SpriteSheet* get_sprite(
        ObjectState state,
        AnimationFrame frame,
        bool special
    ) {
        SpriteKey key{state, frame, special};
        auto it = sprites.find(key);
        return it != sprites.end() ? it->second : nullptr;
    }
};
```
