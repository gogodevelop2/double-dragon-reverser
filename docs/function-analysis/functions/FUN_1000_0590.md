## 🔄 C++ 재구현

```cpp
struct Player {
    int x, y;
    bool alive;
};

struct Camera {
    int x, y;
    int max_scroll_x;
};

struct ViewportBounds {
    int min_x, max_x;
    int min_y, max_y;
};

ViewportBounds calculate_viewport(
    const Player& p1,
    const Player& p2,
    Camera& camera
) {
    // 1. 살아있는 플레이어 좌표 수집
    int x1 = p1.alive ? p1.x : p2.x;
    int y1 = p1.alive ? p1.y : p2.y;

    int x2 = p2.alive ? p2.x : p1.x;
    int y2 = p2.alive ? p2.y : p1.y;

    // 2. Min/Max 계산
    int max_x = std::max(x1, x2);
    int min_x = std::min(x1, x2);

    int max_y = std::max(y1, y2);
    int min_y = std::min(y1, y2);

    // 3. 스크롤 제한 업데이트
    camera.max_scroll_x = std::max(camera.max_scroll_x, max_x);

    // 4. 카메라 상대 좌표로 변환
    return {
        .min_x = min_x - camera.x,
        .max_x = max_x - camera.x,
        .min_y = min_y - camera.y,
        .max_y = max_y - camera.y
    };
}
```
